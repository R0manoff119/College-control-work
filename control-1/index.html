CREATE TABLE Objects ( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(150) NOT NULL, address VARCHAR(200), object_type VARCHAR(100), site_id INT NOT NULL, FOREIGN KEY (site_id) REFERENCES Sites(id) ); CREATE TABLE Contracts ( id INT AUTO_INCREMENT PRIMARY KEY, contract_number VARCHAR(50) NOT NULL, contract_date DATE, customer_id INT NOT NULL, object_id INT NOT NULL, FOREIGN KEY (customer_id) REFERENCES Customers(id), FOREIGN KEY (object_id) REFERENCES Objects(id) ); CREATE TABLE Equipment ( id INT AUTO_INCREMENT PRIMARY KEY, name VARCHAR(100) NOT NULL, equipment_type VARCHAR(100), department_id INT NOT NULL, FOREIGN KEY (department_id) REFERENCES Departments(id) ); CREATE TABLE Equipment_Distribution ( id INT AUTO_INCREMENT PRIMARY KEY, equipment_id INT NOT NULL, object_id INT NOT NULL, FOREIGN KEY (equipment_id) REFERENCES Equipment(id), FOREIGN KEY (object_id) REFERENCES Objects(id) ); CREATE TABLE Brigade_Work ( id INT AUTO_INCREMENT PRIMARY KEY, brigade_id INT NOT NULL, object_id INT NOT NULL, start_date DATE, end_date DATE, FOREIGN KEY (brigade_id) REFERENCES Brigades(id), FOREIGN KEY (object_id) REFERENCES Objects(id) );
