SELECT
`products`.`arcticle`, `products`.`name`, `products`.`unit`,
`products`.`price`, `products`.`discount`, `products`.`amount`,
`products`.`description`, `products`.`photo`,
`categories`.`name` AS 'category',
`producers`.`name` AS 'producer',
`providers`.`name` AS 'provider'
FROM `products`
JOIN `categories` ON `products`.`id_category` = `categories`.`id`
JOIN `producers` ON `products`.`id_producer` = `producers`.`id`
JOIN `providers` ON `products`.`id_provider` = `providers`.`id`
