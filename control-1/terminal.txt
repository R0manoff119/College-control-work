    <h2>Добавить студента</h2>

    <input id="name" placeholder="Имя">
    <input id="surname" placeholder="Фамилия">
    <input id="group" placeholder="Группа">

    <button id="add">Добавить</button>

    <h2>Поиск студента</h2>

    <input id="search" placeholder="Введите имя">

    <h2>Студенты</h2>

    <div id="list"></div>

    <script src="script.js"></script>

let students = JSON.parse(localStorage.getItem("students")) || [];

let name = document.getElementById("name");
let surname = document.getElementById("surname");
let group = document.getElementById("group");
let add = document.getElementById("add");
let search = document.getElementById("search");
let list = document.getElementById("list");

let edit = -1;

function save() {
    localStorage.setItem("students", JSON.stringify(students));
}

function show() {
    list.innerHTML = "";

    students.forEach(function(student, index) {
        if (student.name.toLowerCase().includes(search.value.toLowerCase())) {
            let div = document.createElement("div");

            div.innerHTML = `
                <p>${student.name} ${student.surname}</p>
                <p>Группа: ${student.group}</p>
                <button onclick="change(${index})">Изменить</button>
                <button onclick="remove(${index})">Удалить</button>
            `;

            list.appendChild(div);
        }
    });
}

add.onclick = function() {
    let student = {
        name: name.value,
        surname: surname.value,
        group: group.value
    };

    if (edit === -1) {
        students.push(student);
    } else {
        students[edit] = student;
        edit = -1;
        add.textContent = "Добавить";
    }

    save();
    show();

    name.value = "";
    surname.value = "";
    group.value = "";
};

function change(index) {
    name.value = students[index].name;
    surname.value = students[index].surname;
    group.value = students[index].group;

    edit = index;
    add.textContent = "Сохранить";
}

function remove(index) {
    students.splice(index, 1);

    save();
    show();
}

search.oninput = show;

show();
