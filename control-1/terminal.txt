<input id="city" placeholder="Введите город">
<button onclick="addCity()">Добавить</button>

<p id="error"></p>
<ul id="list"></ul>


let cities = [];

function addCity() {
    let city = document.getElementById("city").value;

    if (cities.includes(city)) {
        document.getElementById("error").innerHTML = "Город уже есть!";
    } else {
        cities.push(city);
        document.getElementById("error").innerHTML = "";
    }

    document.getElementById("list").innerHTML = "";

    for (let i = 0; i < cities.length; i++) {
        document.getElementById("list").innerHTML += "<li>" + cities[i] + "</li>";
    }
}
